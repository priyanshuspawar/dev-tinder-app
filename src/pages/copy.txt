import UnreadMessageInview from "@/components/unread-message";
import { chatinitData, Message } from "@/queries/chatQueries";
import useAuthStore from "@/store/authStore";
import { useNotificationStore } from "@/store/notificationStore";
import useSocket from "@/store/socketStore";
import { groupMessagesByDate } from "@/utils/helpers";
import { Spinner } from "@heroui/react";
import { useQuery } from "@tanstack/react-query";
import axios from "axios";
import clsx from "clsx";
import { Check, CheckCheck, Clock } from "lucide-react";
import { useEffect, useState, useRef } from "react";
import InfiniteScroll from "react-infinite-scroll-component";
import { useParams } from "react-router-dom";

export type responseApi = {
  data: Message[];
  page: number;
  limit: number;
  hasMore: boolean;
};

type SocketMsg = {
  tempId: string;
  _id: string;
  chatId: string;
  senderId: string;
  content: {
    type: string;
    text: string;
  };
  createdAt: string;
  status: "sent" | "delivered" | "read";
};

const Chat = () => {
  const { user } = useAuthStore();
  const { connectionId } = useParams<{ connectionId: string }>();
  const [isFetching, setIsFetching] = useState(false);
  const [chatApiData, setChatApiData] = useState<responseApi | null>(null);
  const [message, setMessage] = useState("");
  const [isSending, setIsSending] = useState(false);
  const [chatError, setChatError] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const fetchingRef = useRef(false); // Prevent multiple simultaneous fetches
  const { newChatMessages } = useNotificationStore();

  const { data: chatData } = useQuery({
    queryKey: ["initilize-chat", connectionId],
    queryFn: () => chatinitData(connectionId!),
    staleTime: 1000 * 60 * 10,
  });
  const fetchMessages = async (
    page: number,
    limit: number,
    lastReceivedId?: string
  ) => {
    if (fetchingRef.current) return; // Prevent concurrent fetches

    try {
      fetchingRef.current = true;
      setIsFetching(true);

      if (chatApiData?.page === page) return;

      const res = await axios.get<responseApi>(
        `http://localhost:5000/chat/${chatData?.data.chatId}?page=${page}&limit=${limit}${lastReceivedId ? `&lastReceivedId=${lastReceivedId}` : ""}`,
        { withCredentials: true }
      );
      return res.data;
    } catch (error) {
      setChatError(
        error instanceof Error
          ? error.message
          : "An error occurred while fetching messages."
      );
    } finally {
      setIsFetching(false);
      fetchingRef.current = false;
    }
  };
  const { socket } = useSocket();
  useEffect(() => {
    if (!chatData || !socket || !chatApiData || !user) {
      return;
    }
    //join room
    socket.emit("joinRoom", chatData.data.chatId);

    //handlers
    const handleReceiveMsg = (msg: SocketMsg) => {
      if (msg.senderId == user._id) return;

      let newData = {
        ...chatApiData,
        data: [
          {
            ...msg,
          },
          ...chatApiData.data,
        ],
      };

      setChatApiData(newData);
      // 🔁 Emit delivered status
      socket.emit("messageReceivedAck", {
        messageId: msg._id,
        chatId: msg.chatId,
        status: "delivered",
      });
    };
    const handleDeliveredAck = ({ messageId }: { messageId: string }) => {
      let index = chatApiData.data.findIndex((val) => val._id === messageId);
      if (index === -1) return;
      let update = chatApiData.data;
      update[index].status = "delivered";
      setChatApiData((prev) => {
        if (!prev) return prev;
        return {
          ...prev,
          data: update,
        };
      });
    };
    socket.on("receiveMessage", handleReceiveMsg);
    socket.on("messageDeliveredAck", handleDeliveredAck);
    return () => {
      socket.off("receiveMessage", handleReceiveMsg);
      socket.off("messageDeliveredAck", handleDeliveredAck);
    };
  }, [chatData, chatApiData]);

  const container = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    if (!connectionId || !chatData) return;
    if (newChatMessages[connectionId]) {
      const lastReceivedId =
        newChatMessages[connectionId][newChatMessages[connectionId].length - 1]
          ._id;

      (async () => {
        const data = await fetchMessages(1, 10, lastReceivedId);
        if (data) {
          setChatApiData(data);
          setIsLoading(false);
        }
      })();
    } else {
      (async () => {
        const data = await fetchMessages(1, 10);
        if (data) {
          setChatApiData(data);
          setIsLoading(false);
        }
      })();
    }
  }, [newChatMessages, chatData, connectionId]);

  if (isLoading || !chatApiData || !chatApiData.data) {
    return (
      <div className="w-full h-full flex flex-col items-center justify-center">
        <Spinner />
      </div>
    );
  }

  const groupedMsgs = groupMessagesByDate(chatApiData.data);

  const socketSend = (text: string, tempId: string, state: Message[]) => {
    if (!socket || !chatApiData || !chatData) return;

    socket.emit(
      "sendMessage",
      {
        tempId,
        roomId: chatData.data.chatId,
        content: {
          type: "text",
          text: text,
        },
      },
      (msg: SocketMsg) => {
        for (let i = 0; i < state.length; i++) {
          if (state[i]._id === msg.tempId) {
            let update = state;
            update[i] = {
              ...msg,
            };
            console.log(update);
            setChatApiData((prev) => {
              if (!prev) return prev;
              return {
                ...prev,
                data: update,
              };
            });
            break;
          }
        }
      }
    );
  };

  return (
    <div className="flex flex-col w-full h-full">
      <div
        id="scrollableDiv"
        // onScroll={(e) => {
        //   console.log(e.currentTarget.scrollTop);
        // }}
        ref={container}
        style={{
          height: 300,
          overflow: "auto",
          display: "flex",
          flexDirection: "column-reverse",
        }}
      >
        <InfiniteScroll
          dataLength={chatApiData.data.length}
          next={async () => {
            // Multiple protection layers
            if (isFetching || fetchingRef.current || !chatApiData.hasMore) {
              return;
            }

            const data = await fetchMessages(chatApiData.page + 1, 10);
            if (data && data.data.length > 0) {
              setChatApiData((prev) => {
                if (!prev) return prev;

                // Check for duplicates
                const existingIds = new Set(prev.data.map((msg) => msg._id));
                const newMessages = data.data.filter(
                  (msg) => !existingIds.has(msg._id)
                );

                if (newMessages.length === 0) return prev; // No new messages

                return {
                  hasMore: data.hasMore,
                  limit: data.limit,
                  page: data.page,
                  data: [...prev.data, ...newMessages],
                };
              });
            }
          }}
          style={{
            display: "flex",
            flexDirection: "column-reverse",
          }}
          inverse={true}
          hasMore={chatApiData.hasMore && !isFetching}
          loader={<h4>Loading...</h4>}
          scrollableTarget="scrollableDiv"
          // Key props to prevent over-triggering
          scrollThreshold={0.7}
          // initialScrollY={-300}
        >
          {Object.keys(groupedMsgs).map((dateKey) => (
            <div key={`group-${dateKey}`} className="mb-4">
              <div className="text-center text-gray-500 text-sm py-2 sticky top-0 bg-white/80 backdrop-blur-sm">
                {dateKey}
              </div>
              {groupedMsgs[dateKey].map((msg, idx) => {
                const showSeparator =
                  msg.status !== "read" &&
                  (idx === 0 ||
                    groupedMsgs[dateKey][idx - 1].status === "read");
                if (msg.status === "read") {
                  return (
                    <div
                      key={msg._id}
                      className="p-2 bg-default-50 rounded-lg mb-2"
                    >
                      <p className="text-sm">{msg.content.text}</p>
                      <span className="text-xs text-gray-400">
                        {new Date(msg.createdAt).toLocaleTimeString()}
                        <StatusIcons status={msg.status} />
                      </span>
                    </div>
                  );
                }

                return (
                  <UnreadMessageInview
                    key={msg._id}
                    containerRef={container}
                    id={msg._id}
                  >
                    {showSeparator && (
                      <>
                        <div className="flex items-center my-2"></div>
                        <div className="flex-grow border-t border-gray-300" />
                        <span className="mx-2 text-xs text-gray-400">
                          Unread
                        </span>
                        <div className="flex-grow border-t border-gray-300" />
                      </>
                    )}
                    <div className="p-2 bg-default-50 rounded-lg mb-2">
                      <p className="text-sm">{msg.content.text}</p>
                      <span className="text-xs text-gray-400">
                        {new Date(msg.createdAt).toLocaleTimeString()}
                        <StatusIcons status={msg.status} />
                      </span>
                    </div>
                  </UnreadMessageInview>
                );
              })}
            </div>
          ))}
        </InfiniteScroll>
      </div>

      <div className="flex gap-2 p-4 border-t">
        <input
          type="text"
          value={message}
          onChange={(e) => setMessage(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === "Enter" && !e.shiftKey) {
              e.preventDefault();
              if (message.trim()) {
                // Trigger send message
                document.querySelector("button")?.click();
              }
            }
          }}
          placeholder="Type a message..."
          disabled={isSending}
          className="flex-1 p-2 border rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
        />
        <button
          onClick={() => {
            if (!message.trim()) return;

            const newMessage = {
              _id: `msg-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
              chatId: "68887212cfdfefd",
              senderId: "54545487878cfcdf",
              content: {
                type: "text",
                text: message,
              },
              createdAt: new Date().toISOString(),
            };
            let newData: { data: Message[] } = {
              data: [newMessage, ...chatApiData.data],
            };

            setChatApiData((prev) => {
              if (!prev) return prev;
              return {
                ...prev,
                data: newData.data,
              };
            });

            socketSend(message, newMessage._id, newData.data);

            setMessage("");
          }}
          disabled={!message.trim() || isSending}
          className="px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600 disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {isSending ? "Sending..." : "Send"}
        </button>
      </div>
    </div>
  );
};

export default Chat;

const StatusIcons = ({
  status,
  className,
}: {
  status?: "sent" | "delivered" | "read";
  className?: string;
}) => {
  switch (status) {
    case "sent":
      return <Check className={clsx(className)} />;
    case "delivered":
      return <CheckCheck className={clsx(className)} />;
    case "read":
      return <CheckCheck className={clsx("text-sky-500", className)} />;
    default:
      return <Clock />;
  }
};
